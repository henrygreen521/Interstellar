const express = require('express'); const http = require('http'); const { Server } = require('socket.io'); const cors = require('cors'); const sqlite3 = require('sqlite3').verbose(); const bcrypt = require('bcryptjs'); const app = express(); app.use(cors()); app.use(express.json({ limit: '10mb' })); // Increased limit to support base64 image profile pictures uploaded from storage const server = http.createServer(app); const io = new Server(server, { cors: { origin: "*", methods: ["GET", "POST"] } }); // --- SQLite Database Setup --- const db = new sqlite3.Database('./chickencoopp.db', (err) => { if (err) console.error('Database opening error: ', err.message); else console.log('Connected to SQLite database.'); }); db.serialize(() => { db.run(`CREATE TABLE IF NOT EXISTS users ( id INTEGER PRIMARY KEY AUTOINCREMENT, username TEXT UNIQUE, password TEXT, pfp TEXT, status_text TEXT DEFAULT '' )`); db.run(`CREATE TABLE IF NOT EXISTS friendships ( id INTEGER PRIMARY KEY AUTOINCREMENT, user_id INTEGER, friend_id INTEGER, status TEXT )`); db.run(`CREATE TABLE IF NOT EXISTS messages ( id INTEGER PRIMARY KEY AUTOINCREMENT, sender_id INTEGER, receiver_id INTEGER, message TEXT, timestamp DATETIME DEFAULT CURRENT_TIMESTAMP )`); db.run(`CREATE TABLE IF NOT EXISTS groups ( id INTEGER PRIMARY KEY AUTOINCREMENT, name TEXT, admin_id INTEGER )`); db.run(`CREATE TABLE IF NOT EXISTS group_members ( group_id INTEGER, user_id INTEGER )`); }); // --- Auth Routes --- app.post('/api/register', async (req, res) => { const { username, password, pfp } = req.body; if (!username || !password) return res.status(400).json({ error: "Username and password required" }); try { const hashedPassword = await bcrypt.hash(password, 10); db.run(`INSERT INTO users (username, password, pfp) VALUES (?, ?, ?)`, [username, hashedPassword, pfp || ''], function(err) { if (err) return res.status(400).json({ error: "Username already taken" }); res.json({ user: { id: this.lastID, username, pfp: pfp || '', status_text: '' } }); }); } catch (e) { res.status(500).json({ error: "Server error" }); } }); app.post('/api/login', (req, res) => { const { username, password } = req.body; if (!username || !password) return res.status(400).json({ error: "Username and password required" }); db.get(`SELECT * FROM users WHERE username = ?`, [username], async (err, user) => { if (err || !user) return res.status(400).json({ error: "Invalid username or password" }); const validPassword = await bcrypt.compare(password, user.password); if (!validPassword) return res.status(400).json({ error: "Invalid username or password" }); res.json({ user: { id: user.id, username: user.username, pfp: user.pfp, status_text: user.status_text } }); }); }); // --- User Profile & Status Routes --- app.post('/api/user/status', (req, res) => { const { userId, status_text } = req.body; db.run(`UPDATE users SET status_text = ? WHERE id = ?`, [status_text, userId], (err) => { if (err) return res.status(500).json({ error: "Database error" }); res.json({ success: true }); }); }); // --- User Search Route --- app.get('/api/users/search', (req, res) => { const query = req.query.q || ''; db.all(`SELECT id, username, pfp, status_text FROM users WHERE username LIKE ?`, [`%${query}%`], (err, rows) => { if (err) return res.status(500).json({ error: "Database error" }); res.json(rows); }); }); // --- Friends Routes --- app.get('/api/friends/:userId', (req, res) => { const userId = req.params.userId; const query = ` SELECT u.id, u.username, u.pfp, u.status_text FROM users u JOIN friendships f ON (u.id = f.friend_id OR u.id = f.user_id) WHERE (f.user_id = ? OR f.friend_id = ?) AND f.status = 'accepted' AND u.id != ? `; db.all(query, [userId, userId, userId], (err, rows) => { if (err) return res.status(500).json({ error: "Database error" }); res.json(rows); }); }); app.get('/api/friends/requests/:userId', (req, res) => { const userId = req.params.userId; const query = ` SELECT u.id, u.username, u.pfp FROM users u JOIN friendships f ON u.id = f.user_id WHERE f.friend_id = ? AND f.status = 'pending' `; db.all(query, [userId], (err, rows) => { if (err) return res.status(500).json({ error: "Database error" }); res.json(rows); }); }); app.post('/api/friends/request', (req, res) => { const { sender_id, receiver_id } = req.body; db.run(`INSERT INTO friendships (user_id, friend_id, status) VALUES (?, ?, 'pending')`, [sender_id, receiver_id], (err) => { if (err) return res.status(500).json({ error: "Database error" }); res.json({ success: true }); }); }); app.post('/api/friends/respond', (req, res) => { const { user_id, sender_id, action } = req.body; if (action === 'accept') { db.run(`UPDATE friendships SET status = 'accepted' WHERE user_id = ? AND friend_id = ?`, [sender_id, user_id], (err) => { if (err) return res.status(500).json({ error: "Database error" }); res.json({ success: true }); }); } else { db.run(`DELETE FROM friendships WHERE user_id = ? AND friend_id = ?`, [sender_id, user_id], (err) => { if (err) return res.status(500).json({ error: "Database error" }); res.json({ success: true }); }); } }); // --- Group Chat Routes --- app.post('/api/groups/create', (req, res) => { const { name, admin_id, member_ids } = req.body; db.run(`INSERT INTO groups (name, admin_id) VALUES (?, ?)`, [name, admin_id], function(err) { if (err) return res.status(500).json({ error: "Database error" }); const groupId = this.lastID; const members = [...(member_ids || []), admin_id]; const stmt = db.prepare(`INSERT INTO group_members (group_id, user_id) VALUES (?, ?)`); members.forEach(userId => stmt.run(groupId, userId)); stmt.finalize(); res.json({ success: true, groupId }); }); }); // --- Messages History Route --- app.get('/api/messages/:user1/:user2', (req, res) => { const { user1, user2 } = req.params; const query = ` SELECT * FROM messages WHERE (sender_id = ? AND receiver_id = ?) OR (sender_id = ? AND receiver_id = ?) ORDER BY timestamp ASC `; db.all(query, [user1, user2, user2, user1], (err, rows) => { if (err) return res.status(500).json({ error: "Database error" }); res.json(rows); }); }); // --- Global Chat History Route --- app.get('/api/messages/global', (req, res) => { const query = ` SELECT m.*, u.username, u.pfp FROM messages m JOIN users u ON m.sender_id = u.id WHERE m.receiver_id = 0 ORDER BY timestamp ASC LIMIT 100 `; db.all(query, [], (err, rows) => { if (err) return res.status(500).json({ error: "Database error" }); res.json(rows); }); }); // --- Socket.io Real-Time Communication --- io.on('connection', (socket) => { console.log('A user connected:', socket.id); socket.on('join_room', (userId) => { socket.userId = userId; socket.join(`user_${userId}`); socket.join('global_room'); // Automatically join global chatroom io.emit('presence_update', { userId, status: 'online' }); }); socket.on('join_group', (groupId) => { socket.join(`group_${groupId}`); }); // --- Whiteboard Room & Events --- socket.on('join_whiteboard', () => { socket.join('whiteboard_room'); }); socket.on('draw_line', (data) => { socket.to('whiteboard_room').emit('draw_line', data); }); socket.on('clear_whiteboard', () => { socket.to('whiteboard_room').emit('clear_whiteboard'); }); // --- Game Invite Events --- socket.on('send_game_invite', (data) => { const { sender_id, receiver_id, game_name } = data; io.to(`user_${receiver_id}`).emit('receive_game_invite', { sender_id, game_name }); }); socket.on('respond_game_invite', (data) => { const { sender_id, receiver_id, accept } = data; io.to(`user_${sender_id}`).emit('game_invite_response', { responder_id: receiver_id, accept }); }); // --- WebRTC Signaling Events --- socket.on('call-ring', (data) => { // Direct forwarding for the incoming call banner if (data && data.to) { io.to(`user_${data.to}`).emit('call-ring', data); } }); socket.on('cancel-call', (data) => { if (data && data.to) { io.to(`user_${data.to}`).emit('call-cancelled', data); } }); socket.on('accept-call', (data) => { if (data && data.to) { io.to(`user_${data.to}`).emit('call-accepted', data); } }); socket.on('reject-call', (data) => { if (data && data.to) { io.to(`user_${data.to}`).emit('call-rejected', data); } }); socket.on('call_user', (data) => { io.to(`user_${data.userToCall}`).emit('incoming_call', { signal: data.signalData, from: data.from, name: data.name }); }); socket.on('answer_call', (data) => { io.to(`user_${data.to}`).emit('call_accepted', data.signal); }); socket.on('call-offer', (data) => { if (data && data.to) { io.to(`user_${data.to}`).emit('call-offer', { offer: data.offer }); } }); socket.on('call-answer', (data) => { if (data && data.to) { io.to(`user_${data.to}`).emit('call-answer', { answer: data.answer }); } }); socket.on('ice_candidate', (data) => { io.to(`user_${data.to}`).emit('ice_candidate', data.candidate); }); socket.on('ice-candidate', (data) => { if (data && data.to) { io.to(`user_${data.to}`).emit('ice-candidate', { candidate: data.candidate }); } }); socket.on('end_call', (data) => { io.to(`user_${data.to}`).emit('call_ended'); }); socket.on('end-call', (data) => { if (data && data.to) { io.to(`user_${data.to}`).emit('call-ended'); } }); socket.on('send_message', (data) => { const { sender_id, receiver_id, message } = data; db.run( `INSERT INTO messages (sender_id, receiver_id, message) VALUES (?, ?, ?)`, [sender_id, receiver_id, message], function(err) { if (!err) { const messageData = { id: this.lastID, sender_id, receiver_id, message }; io.to(`user_${receiver_id}`).emit('receive_message', messageData); io.to(`user_${sender_id}`).emit('receive_message', messageData); } } ); }); // Global chat message broadcasting (receiver_id = 0 represents global room) socket.on('send_global_message', (data) => { const { sender_id, message, username, pfp } = data; db.run( `INSERT INTO messages (sender_id, receiver_id, message) VALUES (?, 0, ?)`, [sender_id, message], function(err) { if (!err) { io.to('global_room').emit('receive_global_message', { id: this.lastID, sender_id, message, username, pfp }); } } ); }); socket.on('send_group_message', (data) => { const { group_id, sender_id, message } = data; db.run( `INSERT INTO messages (sender_id, receiver_id, message) VALUES (?, ?, ?)`, [sender_id, -group_id, message], function(err) { if (!err) { io.to(`group_${group_id}`).emit('receive_group_message', { id: this.lastID, group_id, sender_id, message }); } } ); }); // Real-time typing indicators socket.on('typing', (data) => { io.to(`user_${data.receiver_id}`).emit('display_typing', { sender_id: data.sender_id }); }); socket.on('stop_typing', (data) => { io.to(`user_${data.receiver_id}`).emit('hide_typing', { sender_id: data.sender_id }); }); socket.on('disconnect', () => { if (socket.userId) { io.emit('presence_update', { userId: socket.userId, status: 'offline' }); } console.log('User disconnected:', socket.id); }); }); const PORT = process.env.PORT || 3000; server.listen(PORT, () => { console.log(`Server running on port ${PORT}`); });
